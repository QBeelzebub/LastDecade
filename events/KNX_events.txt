country_event = {
	id = knx.8
	title = knx.8.t
	desc = knx.8.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	option = {
		name = knx.8.a
		ai_chance = { base = 100 }
		369 = {
			if = {
				limit = {
					NOT = { is_owned_by = KNX }
					NOT = { is_owned_by = EFH }
					NOT = { owner = { is_in_faction_with = EFH } }
				}
				owner = { country_event = { id = knx.9 } }
			}
			transfer_state_to = KNX
			add_core_of = KNX
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = knx_root_stronghold } } }
				add_dynamic_modifier = { modifier = knx_root_stronghold }
			}
		}
		if = {
			limit = { KNX = { is_government_in_exile = yes } }
			EFH = { end_exile = KNX }
		}
		KNX = {
			set_capital = { state = 369 }
			if = {
				limit = { NOT = { has_idea = KNX_anomalous_infestation } }
				add_ideas = KNX_anomalous_infestation
			}
			if = {
				limit = { NOT = { has_idea = KNX_civilization_project } }
				add_ideas = KNX_civilization_project
			}
			if = {
				limit = { NOT = { has_idea = KNX_dug_in_homeland } }
				add_ideas = KNX_dug_in_homeland
			}
			if = {
				limit = { has_army_size = { size < 10 } }
				add_manpower = 50000
				add_equipment_to_stockpile = { type = rupter_equipment_1 amount = 6000 }
				load_oob = "KNX_initial_army"
			}
			KNX_grow_hive_forts_effect = yes
		}
		clr_global_flag = KNX_in_exile
		clr_global_flag = KNX_return_pending
	}
}
