		ai_chance = {
			base = 30
			modifier = { factor = 0 is_historical_focus_on = yes }

		CHN = {
			clr_country_flag = CHN_efh_offer_pending

		}
	}
}

country_event = {
	id = chn_diplomacy.2
	title = chn_diplomacy.2.t
	desc = chn_diplomacy.2.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	trigger = { tag = USA country_exists = CHN }
	option = {
		name = chn_diplomacy.2.a
		ai_chance = { base = 70 }
		if = {
			limit = {
				country_exists = CHN
				is_subject = no
				has_capitulated = no
				NOT = { has_war_with = CHN }
				CHN = {
					is_subject = no
					has_capitulated = no
					NOT = { is_in_faction_with = EFH }
					NOT = { has_country_flag = CHN_efh_aligned }
					NOT = { has_country_flag = CHN_efh_offer_pending }
					NOT = { has_country_flag = CHN_usa_aligned }
				}
			}
			CHN = {
				set_country_flag = CHN_usa_aligned
				clr_country_flag = CHN_efh_offer_pending
				clr_country_flag = CHN_canal_initiative
				if = {
					limit = { NOT = { has_idea = CHN_usa_partnership } }
					add_ideas = CHN_usa_partnership
				}
			}
			if = {
				limit = { CHN = { NOT = { is_guaranteed_by = USA } } }
				give_guarantee = CHN
			}
			USA_trust_very_low_gain_effect = yes
			if = {
				limit = { country_exists = EFH }
				EFH = {
					clr_country_flag = EFH_china_leaning_efh
					set_country_flag = EFH_china_neutral
				}
			}
		}
		CHN = {
			clr_country_flag = CHN_usa_offer_pending
			country_event = { id = chn_diplomacy.5 days = 1 }
		}
	}
	option = {
		name = chn_diplomacy.2.b
		ai_chance = { base = 30 }
		CHN = {
			clr_country_flag = CHN_usa_offer_pending
			country_event = { id = chn_diplomacy.5 days = 1 }
		}
	}
}
