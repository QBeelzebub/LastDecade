@IRS_tromos_auto3_days = 63
@IRS_tromos_auto3_flag = 64

country_event = {
	id = irs_tromos.6
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = IRS_tromos_auto3_in_progress
		NOT = { has_completed_focus = IRS_tromos_auto_3 }
	}

	immediate = {
		complete_national_focus = IRS_tromos_auto_3
	}
}

country_event = {
	id = irs_tromos.5
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_completed_focus = IRS_tromos_auto_2
		NOT = { has_completed_focus = IRS_tromos_auto_3 }
		NOT = { has_country_flag = IRS_tromos_auto3_in_progress }
	}

	immediate = {
		activate_shine_on_focus = IRS_tromos_auto_3
		set_country_flag = {
			flag = IRS_tromos_auto3_in_progress
			days = @IRS_tromos_auto3_flag
			value = 1
		}
		country_event = { id = irs_tromos.6 days = @IRS_tromos_auto3_days }
	}
}
