country_event = {
	id = jap.1
	title = jap.1.t
	desc = jap.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	option = {
		name = jap.1.a
		if = {
			limit = { has_idea = JAP_island_economy }
			swap_ideas = { remove_idea = JAP_island_economy add_idea = JAP_uninsured_sea }
		}
		if = {
			limit = {
				NOT = { has_idea = JAP_island_economy }
				NOT = { has_idea = JAP_uninsured_sea }
			}
			add_ideas = JAP_uninsured_sea
		}
	}
}
