country_event = {
	id = irs_vitreous.6
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = IRS_vitreous_auto3_in_progress
		NOT = { has_completed_focus = IRS_vitreous_auto_3 }
	}

	immediate = {
		complete_national_focus = IRS_vitreous_auto_3
	}
}

		activate_shine_on_focus = IRS_vitreous_auto_3

}
