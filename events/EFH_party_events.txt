add_namespace = efh_party

country_event = {
	id = efh_party.1
	title = efh_party.1.t
	desc = efh_party.1.d
	hidden = yes
	is_triggered_only = yes

	trigger = {
		original_tag = EFH
		has_country_flag = EFH_party_system_initialized
		NOT = { has_country_flag = EFH_old_nations_released }
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = EFH_federal_election_in_progress
			EFH_party_clamp_support_effect = yes
			if = {
				limit = {
					check_variable = { var = EFH_party_astra value = EFH_party_volt compare = greater_than_or_equals }
					check_variable = { var = EFH_party_astra value = EFH_party_ave compare = greater_than_or_equals }
					check_variable = { var = EFH_party_astra value = EFH_party_continuity compare = greater_than_or_equals }
					check_variable = { var = EFH_party_astra value = EFH_party_social_green compare = greater_than_or_equals }
					check_variable = { var = EFH_party_astra value = EFH_party_sovereigntist compare = greater_than_or_equals }
				}
				EFH_party_apply_astra_win_effect = yes
				news_event = { id = efh_party.2 days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { var = EFH_party_volt value = EFH_party_ave compare = greater_than_or_equals }
					check_variable = { var = EFH_party_volt value = EFH_party_continuity compare = greater_than_or_equals }
					check_variable = { var = EFH_party_volt value = EFH_party_social_green compare = greater_than_or_equals }
					check_variable = { var = EFH_party_volt value = EFH_party_sovereigntist compare = greater_than_or_equals }
				}
				EFH_party_apply_volt_win_effect = yes
				news_event = { id = efh_party.3 days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { var = EFH_party_ave value = EFH_party_continuity compare = greater_than_or_equals }
					check_variable = { var = EFH_party_ave value = EFH_party_social_green compare = greater_than_or_equals }
					check_variable = { var = EFH_party_ave value = EFH_party_sovereigntist compare = greater_than_or_equals }
				}
				EFH_party_apply_ave_win_effect = yes
				news_event = { id = efh_party.4 days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { var = EFH_party_continuity value = EFH_party_social_green compare = greater_than_or_equals }
					check_variable = { var = EFH_party_continuity value = EFH_party_sovereigntist compare = greater_than_or_equals }
				}
				EFH_party_apply_continuity_win_effect = yes
				news_event = { id = efh_party.5 days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { var = EFH_party_social_green value = EFH_party_sovereigntist compare = greater_than_or_equals }
				}
				EFH_party_apply_social_green_win_effect = yes
				news_event = { id = efh_party.6 days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { var = EFH_party_sovereigntist value = EFH_party_social_green compare = greater_than }
				}
				EFH_party_apply_sovereigntist_win_effect = yes
				news_event = { id = efh_party.7 days = 1 }
			}
			set_country_flag = EFH_regular_federal_elections_active
		}
	}
}

country_event = {
	id = efh_party.40
	hidden = yes
	is_triggered_only = yes

	trigger = {
		original_tag = EFH
		has_country_flag = EFH_party_system_initialized
		NOT = { has_country_flag = EFH_old_nations_released }
	}

	immediate = {
		if = {
			limit = { has_stability < 0.40 }
			add_to_variable = { var = EFH_separatist_pressure value = 1 }
		}
		if = {
			limit = { has_war = yes }
			add_to_variable = { var = EFH_separatist_pressure value = 1 }
		}
		if = {
			limit = { has_country_flag = EFH_party_sovereigntist_incumbent }
			add_to_variable = { var = EFH_separatist_pressure value = 1 }
		}
		EFH_party_clamp_support_effect = yes
		EFH_party_check_separatist_thresholds_effect = yes
		country_event = { id = efh_party.40 days = 30 }
	}
}
