country_event = {
	id = ldwar.30
	title = ldwar.30.t
	desc = ldwar.30.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	option = {
		name = ldwar.30.a
		ai_chance = {
			base = 50
			modifier = { factor = 2 check_variable = { EFH_USA_escalation > 49 } }
			modifier = { factor = 0.2 check_variable = { EFH_USA_escalation < 25 } }
		}
		declare_war_on = {
			target = USA
			type = annex_everything
		}
		add_ideas = LD_joint_offensive
		set_global_flag = LD_second_atlantic_war_open
		if = {
			limit = {
				country_exists = KNX
				KNX = { NOT = { has_war_with = USA } }
			}
			KNX = {
				declare_war_on = {
					target = USA
					type = annex_everything
				}
				add_ideas = LD_joint_offensive
			}
		}
		USA = { country_event = { id = ldwar.33 days = 1 } }
		news_event = { id = ldwar.34 days = 1 }
	}

	option = {
		name = ldwar.30.b
		ai_chance = { base = 50 }
		add_political_power = 50
		set_country_flag = { flag = LD_second_front_cd value = 1 days = 180 }
		clr_country_flag = LD_second_front_offered
	}
}
