irs_assimilation = {
	irs_absorb_state = {
		icon = GFX_decision_generic_operation

		state_target = yes
		on_map_mode = map_and_decisions_view

		highlight_states = {
			highlight_states_trigger = {
				is_owned_by = IRS
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		target_root_trigger = {
			tag = IRS
			any_owned_state = {
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		target_trigger = {
			FROM = {
				is_owned_by = IRS
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		visible = {
			tag = IRS
		}

		available = {
			tag = IRS
			FROM = {
				is_owned_by = IRS
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		cancel_trigger = {
			FROM = {
				OR = {
					NOT = { is_owned_by = IRS }
					check_variable = {
						var = iris_infestation_level
						value = 50
						compare = less_than
					}
				}
			}
		}

		cost = 100
		days_remove = 14

		remove_effect = {
			FROM = {
				add_core_of = IRS
				add_dynamic_modifier = { modifier = irs_consumed_state }
				set_state_flag = irs_state_consumed
				remove_building = { type = industrial_complex level = 20 }
				remove_building = { type = arms_factory level = 20 }
				remove_building = { type = infrastructure level = 5 }
				remove_building = { type = dockyard level = 20 }
				remove_building = { type = synthetic_refinery level = 3 }
				remove_building = { type = fuel_silo level = 20 }
				remove_building = { type = radar_station level = 6 }
				remove_building = { type = rocket_site level = 3 }
				remove_building = { type = energy_infrastructure level = 1 }
				remove_building = { type = industrial_infrastructure level = 1 }
				add_building_construction = {
					type = beckon_cluster
					level = 1
				}
			}
			set_variable = { var = IRS_ep_gain_tmp value = 25 }
			IRS_grant_scaled_ep = yes
		}

		ai_will_do = {
			base = 800
		}
	}

}
