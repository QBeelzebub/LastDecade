irs_assimilation = {
	irs_absorb_state = {
		icon = GFX_decision_generic_operation

		state_target = yes
		on_map_mode = map_and_decisions_view

		highlight_states = {
			highlight_states_trigger = {
				is_owned_by = ROOT
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		target_root_trigger = {
			tag = IRS
			any_owned_state = {
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		target_trigger = {
			FROM = {
				is_owned_by = ROOT
				is_controlled_by = ROOT
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		visible = {
			tag = IRS
		}

		available = {
			tag = IRS
			FROM = {
				is_owned_by = ROOT
				is_controlled_by = ROOT
				check_variable = {
					var = iris_infestation_level
					value = 99
					compare = greater_than
				}
				NOT = { has_dynamic_modifier = { modifier = irs_consumed_state } }
				NOT = { has_dynamic_modifier = { modifier = knx_consumed_state } }
			}
		}

		cancel_trigger = {
			FROM = {
				OR = {
					NOT = { is_owned_by = ROOT }
					check_variable = {
						var = iris_infestation_level
						value = 50
						compare = less_than
					}
				}
			}
		}

		cost = 75
		days_remove = 14

		remove_effect = {
			FROM = { SRP_absorb_state_for_IRS = yes }
			custom_effect_tooltip = IRS_ep_gain_25_tt
		}

		ai_will_do = {
			base = 8
		}
	}

}
