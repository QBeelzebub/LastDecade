srp_turret_deployment = {
	SRP_deploy_laser_turret = {
		icon = GFX_decision_generic_industry

		state_target = yes
		on_map_mode = map_and_decisions_view

		highlight_states = {
			highlight_states_trigger = {
				is_controlled_by = ROOT
				srp_laser_turret < 1
			}
		}

		target_root_trigger = {
			NOT = { is_full_parasite = yes }
			has_tech = srp_laser_turret_tech
			any_controlled_state = {
				srp_laser_turret < 1
			}
		}

		target_trigger = {
			FROM = {
				is_controlled_by = ROOT
				srp_laser_turret < 1
			}
		}

		visible = {
			NOT = { is_full_parasite = yes }
			has_tech = srp_laser_turret_tech
		}

		available = {
			NOT = { is_full_parasite = yes }
			has_tech = srp_laser_turret_tech
			FROM = {
				is_controlled_by = ROOT
				srp_laser_turret < 1
			}
		}

		cancel_trigger = {
			FROM = {
				OR = {
					NOT = { is_controlled_by = ROOT }
					srp_laser_turret > 0
				}
			}
		}

		cost = 25
		days_remove = 7

		modifier = {
			political_power_gain = -0.05
		}

		remove_effect = {
			FROM = {
				add_building_construction = {
					type = srp_laser_turret
					level = 1
					instant_build = yes
				}
			}
		}

		ai_will_do = { base = 15 }
	}
}
