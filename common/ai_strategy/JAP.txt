
JAP_china_wary = {
	allowed = { original_tag = JAP }
	enable = {
		country_exists = CHN
		NOT = { has_war_with = CHN }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = befriend id = CHN value = 20 }
	ai_strategy = { type = contain  id = CHN value = 40 }
	ai_strategy = { type = alliance id = CHN value = -300 }
	ai_strategy = { type = conquer  id = CHN value = -100 }
}

	allowed = { original_tag = JAP }

		country_exists = CHN

	ai_strategy = { type = befriend id = CHN value = 300 }

}
