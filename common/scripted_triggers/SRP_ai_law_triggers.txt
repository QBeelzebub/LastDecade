	OR = {
		has_war = yes
		has_global_flag = parasites_emerged
		threat > 0.25
		AND = {
			OR = { tag = EFH tag = USA }
			check_variable = { EFH_USA_escalation > 64 }
		}
	}
}

	has_war = yes

		surrender_progress > 0.25

				has_idea = srp_total_supply_lockdown
				has_idea = srp_desperate_enlistment
			}
		}
	}
}
