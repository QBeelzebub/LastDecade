on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = {
					date > 2023.2.24
					NOT = { has_global_flag = KNX_silent_emergence_fired }
					369 = { NOT = { is_owned_by = KNX } }
				}
				set_global_flag = KNX_silent_emergence_fired
				369 = {
					owner = { country_event = { id = knx.1 days = 1 } }
				}
			}
		}
	}

	on_monthly_KNX = {
		effect = {
			KNX_grow_hive_forts_effect = yes
			KNX_hive_state_development_effect = yes
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					tag = KNX
					is_in_faction_with = EFH
					NOT = { has_global_flag = KNX_in_exile }
					country_exists = EFH
					EFH = { has_capitulated = no }
				}
				country_event = { id = knx.7 }
			}
		}
	}

	on_monthly_USA = {
		effect = {
			369 = {
				if = {
					limit = {
						has_dynamic_modifier = { modifier = knx_root_stronghold }
						NOT = { is_controlled_by = KNX }
					}
					remove_dynamic_modifier = { modifier = knx_root_stronghold }
				}
			}
		}
	}

	on_monthly_EFH = {
		effect = {
			if = {
				limit = {
					has_global_flag = KNX_in_exile
					NOT = { has_global_flag = KNX_return_pending }
					369 = {
						NOT = { is_controlled_by = KNX }
						NOT = { controller = { is_full_parasite = yes } }
						OR = {
							is_controlled_by = EFH
							controller = { is_in_faction_with = EFH }
							AND = {
								NOT = { has_global_flag = KNX_exile_first_year }
								check_variable = { iris_infestation_level > 49 }
							}
						}
					}
				}
				set_global_flag = KNX_return_pending
				country_event = { id = knx.8 days = 1 }
			}
		}
	}
}
