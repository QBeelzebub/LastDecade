on_actions = {

    on_startup = {
        effect = {
            IRS = { set_variable = { var = IRS_milestone_50_count value = 0 } }
            every_country = {
                limit = {
                    exists = yes
                    NOT = { is_full_parasite = yes }
                }
                set_variable = { var = public_panic value = 20 }
                set_variable = { var = public_panic_last value = 20 }
                SRP_public_panic_own_floor_effect = yes
                SRP_recalculate_public_panic_effect = yes
            }
        }
    }

    on_daily = {
        effect = {
            if = {
                limit = {
                    exists = yes
                    NOT = { is_full_parasite = yes }
                }
                SRP_recalculate_public_panic_effect = yes
            }
        }
    }

    on_monthly = {
        effect = {
            if = {
                limit = {
                    exists = yes
                    NOT = { is_full_parasite = yes }
                }
                SRP_public_panic_own_floor_effect = yes
                if = {
                    limit = { check_variable = { public_panic > 0 } }
                    add_to_variable = { var = public_panic value = -10 }
                }
                SRP_recalculate_public_panic_effect = yes
            }
        }
    }
}
