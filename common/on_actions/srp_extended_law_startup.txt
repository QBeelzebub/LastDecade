on_actions = {

	on_startup = {
		effect = {

			every_country = {
				limit = {
					exists = yes
					NOT = { is_full_parasite = yes }
				}
				add_ideas = srp_civil_society_intact
				add_ideas = srp_laissez_faire
				set_technology = { human_industry_access = 1 popup = no }
				set_technology = { srp_ifv_chassis_tech = 1 popup = no }
				set_technology = { srp_wheeled_ifv_1 = 1 popup = no }
				set_technology = { srp_atgm_1 = 1 popup = no }
			}

			every_country = {
				limit = {
					exists = yes
					is_parasite_nation = yes
				}
				set_technology = { parasite_equipment_access = 1 popup = no }
				set_technology = { parasite_special_project_access = 1 popup = no }
			}

			every_country = {
				limit = { tag = EFH exists = yes }
				remove_ideas = srp_civil_society_intact
				add_ideas = srp_emergency_legislation
				remove_ideas = srp_laissez_faire
				add_ideas = srp_social_democracy
			}

			every_country = {
				limit = { tag = EFH exists = yes }
				set_variable = { var = SRP_nuclear_warheads value = 40 }
			}
			every_country = {
				limit = { tag = USA exists = yes }
				set_variable = { var = SRP_nuclear_warheads value = 30 }
			}
		}
	}

	on_weekly = {
		effect = {
			every_country = {
				limit = {
					exists = yes
					is_parasite_nation = yes
					NOT = { has_tech = parasite_equipment_access }
				}
				set_technology = { parasite_equipment_access = 1 popup = no }
			}
		}
	}
}
