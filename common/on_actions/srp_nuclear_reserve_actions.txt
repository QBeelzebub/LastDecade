on_actions = {

	on_monthly_EFH = {
		effect = {
			if = {
				limit = { check_variable = { var = global.srp_nuclear_mode value = 2 compare = equals } }

				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 29 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -30 }
					add_to_variable = { var = SRP_nuclear_warheads value = 30 }
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 9 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -10 }
					add_to_variable = { var = SRP_nuclear_warheads value = 10 }
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 4 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -5 }
					add_to_variable = { var = SRP_nuclear_warheads value = 5 }
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 0 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -1 }
					add_to_variable = { var = SRP_nuclear_warheads value = 1 }
				}
				clamp_variable = { var = SRP_nuclear_warheads min = 0 max = 999 }

				if = {
					limit = {
						check_variable = { SRP_nuclear_warheads < 8 }
						NOT = { has_idea = srp_critical_nuclear_reserve }
					}
					add_ideas = srp_critical_nuclear_reserve
					country_event = { id = srp_nuclear.1 days = 1 }
				}
				if = {
					limit = {
						check_variable = { SRP_nuclear_warheads > 15 }
						has_idea = srp_critical_nuclear_reserve
					}
					remove_ideas = srp_critical_nuclear_reserve
				}
			}
			else = {
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 29 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -30 }
					add_nuclear_bombs = 30
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 9 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -10 }
					add_nuclear_bombs = 10
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 4 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -5 }
					add_nuclear_bombs = 5
				}
				if = {
					limit = { has_equipment = { srp_warhead_component_1 > 0 } }
					add_equipment_to_stockpile = { type = srp_warhead_component_1 amount = -1 }
					add_nuclear_bombs = 1
				}
			}
		}
	}
