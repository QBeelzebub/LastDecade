on_actions = {
	on_monthly_JAP = {
		effect = {

			if = {
				limit = {
					date > 2021.3.1
					has_global_flag = parasites_emerged
					NOT = { has_country_flag = JAP_sea_lanes_closed }
				}
				set_country_flag = JAP_sea_lanes_closed
				country_event = { id = jap.1 days = 1 }
			}

			if = {
				limit = {
					country_exists = EFH
					country_exists = USA
					EFH = { has_war_with = USA }
					NOT = { has_global_flag = LD_first_atlantic_war_ended }
					NOT = { has_global_flag = LD_second_atlantic_war_open }
					NOT = { has_country_flag = JAP_atlantic_answered_1 }
					NOT = { has_war_with = EFH }
					NOT = { has_war_with = USA }
					is_subject = no
					has_capitulated = no
				}
				set_country_flag = JAP_atlantic_answered_1
				country_event = { id = jap.2 days = 2 }
			}

			if = {
				limit = {
					country_exists = EFH
					country_exists = USA
					has_global_flag = LD_second_atlantic_war_open
					EFH = { has_war_with = USA }
					NOT = { has_country_flag = JAP_atlantic_answered_2 }
					NOT = { has_war_with = EFH }
					NOT = { has_war_with = USA }
					is_subject = no
					has_capitulated = no
				}
				set_country_flag = JAP_atlantic_answered_2
				country_event = { id = jap.9 days = 2 }
			}

			if = {
				limit = {
					has_global_flag = LD_first_atlantic_war_ended
					NOT = { has_global_flag = LD_second_atlantic_war_open }
					NOT = { has_country_flag = JAP_truce_cd }
					OR = {
						AND = { has_war_with = EFH country_exists = USA NOT = { USA = { has_war_with = EFH } } }
						AND = { has_war_with = USA country_exists = EFH NOT = { EFH = { has_war_with = USA } } }
					}
				}
				set_country_flag = { flag = JAP_truce_cd value = 1 days = 60 }
				country_event = { id = jap.6 days = 1 }
			}

			if = {
				limit = {
					country_exists = CHN
					NOT = { has_war_with = CHN }
					NOT = { has_country_flag = JAP_china_war_declined }
					NOT = { has_country_flag = JAP_asian_theatre_pending }
					is_subject = no
					has_capitulated = no
					CHN = { is_subject = no has_capitulated = no }
					OR = {
						AND = { has_war_with = EFH CHN = { has_war_with = USA } }
						AND = { has_war_with = USA CHN = { has_war_with = EFH } }
					}
				}
				set_country_flag = { flag = JAP_asian_theatre_pending value = 1 days = 30 }
				country_event = { id = jap.3 days = 1 }
			}

			if = {
				limit = {
					has_global_flag = LD_asian_theatre
					has_war_with = CHN
					NOT = {
						OR = {
							AND = { has_war_with = EFH CHN = { has_war_with = USA } }
							AND = { has_war_with = USA CHN = { has_war_with = EFH } }
						}
					}
				}
				country_event = { id = jap.4 days = 1 }
			}

			if = {
				limit = {
					country_exists = CHN
					NOT = { has_war_with = CHN }
					NOT = { has_country_flag = JAP_common_cause_cd }
					OR = {
						AND = { has_war_with = USA CHN = { has_war_with = USA } }
						AND = { has_war_with = EFH CHN = { has_war_with = EFH } }
					}
				}
				set_country_flag = { flag = JAP_common_cause_cd value = 1 days = 730 }
				country_event = { id = jap.5 days = 1 }
			}

			if = {
				limit = {
					has_idea = JAP_armed_neutrality
					OR = { has_war_with = EFH has_war_with = USA }
				}
				remove_ideas = JAP_armed_neutrality
			}
		}
	}
}
