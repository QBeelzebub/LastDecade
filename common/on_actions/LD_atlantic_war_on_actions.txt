on_actions = {
	on_monthly_EFH = {
		effect = {

			if = {
				limit = {
					date > 2020.6.1
					has_country_flag = EFH_china_trade_active
					country_exists = USA
					NOT = { has_war_with = USA }
					NOT = { has_global_flag = LD_canal_china_fired }
					685 = { is_owned_by = USA }
				}
				set_global_flag = LD_canal_china_fired
				country_event = { id = ldwar.1 days = 3 }
				USA = { country_event = { id = ldwar.2 days = 6 } }
			}

			if = {
				limit = {
					date > 2021.9.1
					country_exists = USA
					NOT = { has_war_with = USA }
					NOT = { has_global_flag = LD_second_summit_convened }
				}
				set_global_flag = LD_second_summit_convened
				news_event = { id = efh.4 days = 5 }
			}

			if = {
				limit = {
					has_war_with = USA
					NOT = { has_country_flag = LD_atlantic_war_1_active }
					NOT = { has_global_flag = LD_first_atlantic_war_ended }
					NOT = { has_global_flag = LD_second_atlantic_war_open }
				}
				set_country_flag = LD_atlantic_war_1_active
				set_country_flag = { flag = LD_white_peace_cd value = 1 days = 120 }
				if = {
					limit = {
						country_exists = RSF
						NOT = { has_war_with = RSF }
						RSF = { NOT = { has_completed_focus = RSF_the_fall } }
					}
					RSF = { country_event = { id = ldwar.10 days = 12 } }
				}
			}

			if = {
				limit = {
					has_war_with = USA
					has_country_flag = LD_atlantic_war_1_active
					has_global_flag = parasites_emerged
					NOT = { has_country_flag = LD_white_peace_cd }
					NOT = { has_global_flag = LD_first_atlantic_war_ended }
				}
				set_country_flag = { flag = LD_white_peace_cd value = 1 days = 120 }
				country_event = { id = ldwar.20 days = 4 }
			}

			if = {
				limit = {
					country_exists = KNX
					country_exists = USA
					KNX = { has_country_flag = KNX_efh_strategic_partner }
					NOT = { has_war_with = USA }
					NOT = { has_country_flag = LD_second_front_offered }
					NOT = { has_country_flag = LD_second_front_cd }
					NOT = { has_global_flag = LD_second_atlantic_war_open }
				}
				set_country_flag = LD_second_front_offered
				country_event = { id = ldwar.30 days = 7 }
			}

			if = {
				limit = {
					country_exists = KNX
					country_exists = USA
					KNX = { has_country_flag = KNX_first_accord_signed }
					USA = { has_war_with = KNX }
					NOT = { has_war_with = USA }
					NOT = { has_country_flag = LD_knox_intervention_offered }
					NOT = { has_global_flag = LD_second_atlantic_war_open }
				}
				set_country_flag = LD_knox_intervention_offered
				country_event = { id = ldwar.31 days = 3 }
			}
		}
	}
}
