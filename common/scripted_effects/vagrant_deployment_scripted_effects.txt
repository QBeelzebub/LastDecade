SRP_vagrant_1_advisory_duty_effect = {
	if = {
		limit = { has_country_flag = KNX_vagrant_1_field_duty }
		add_equipment_to_stockpile = { type = vagrant_1_core amount = -1 }
	}
	set_country_flag = KNX_vagrant_1_advisory_duty
	clr_country_flag = KNX_vagrant_1_field_duty
	activate_advisor = KNX_vagrant_one_advisor
	activate_advisor = KNX_vagrant_one_b_advisor
}

SRP_vagrant_1_field_duty_effect = {
	set_country_flag = KNX_vagrant_1_field_duty
	clr_country_flag = KNX_vagrant_1_advisory_duty
	remove_ideas = KNX_vagrant_one_advisor
	remove_ideas = KNX_vagrant_one_b_advisor
	if = {
		limit = {
			NOT = {
				has_equipment = {
					vagrant_1_core = 1
				}
			}
		}
		add_equipment_to_stockpile = { type = vagrant_1_core_1 amount = 1 producer = KNX }
	}
}
