

SRP_vagrant_1_advisory_duty_effect = {
	set_variable = { var = SRP_vagrant_1_core_delta value = 0 }
	subtract_from_variable = { var = SRP_vagrant_1_core_delta value = num_equipment@vagrant_1_core }
	add_equipment_to_stockpile = { type = vagrant_1_core_1 amount = SRP_vagrant_1_core_delta producer = KNX }
	set_country_flag = KNX_vagrant_1_advisory_duty
	clr_country_flag = KNX_vagrant_1_field_duty
	if = {
		limit = { NOT = { has_country_flag = KNX_vagrant_1_copy_b } }
		KNX_vagrant_one = { remove_unit_leader_role = yes }
		activate_advisor = KNX_vagrant_one_advisor
	}
	if = {
		limit = { has_country_flag = KNX_vagrant_1_copy_b }
		KNX_vagrant_one_reborn = { remove_unit_leader_role = yes }
		activate_advisor = KNX_vagrant_one_b_advisor
	}
}

SRP_vagrant_1_field_duty_effect = {
	set_country_flag = KNX_vagrant_1_field_duty
	clr_country_flag = KNX_vagrant_1_advisory_duty
	remove_ideas = KNX_vagrant_one_advisor
	remove_ideas = KNX_vagrant_one_b_advisor
	if = {
		limit = { NOT = { has_country_flag = KNX_vagrant_1_copy_b } }
		add_corps_commander_role = {
			character = KNX_vagrant_one
			traits = { career_officer trait_engineer }
			skill = 5
			attack_skill = 4
			defense_skill = 6
			planning_skill = 4
			logistics_skill = 5
		}
	}
	if = {
		limit = { has_country_flag = KNX_vagrant_1_copy_b }
		add_corps_commander_role = {
			character = KNX_vagrant_one_reborn
			traits = { career_officer trait_engineer }
			skill = 5
			attack_skill = 4
			defense_skill = 6
			planning_skill = 4
			logistics_skill = 5
		}
	}
	set_variable = { var = SRP_vagrant_1_core_delta value = 1 }
	subtract_from_variable = { var = SRP_vagrant_1_core_delta value = num_equipment@vagrant_1_core }
	add_equipment_to_stockpile = { type = vagrant_1_core_1 amount = SRP_vagrant_1_core_delta producer = KNX }
}
