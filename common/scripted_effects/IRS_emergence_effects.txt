
SRP_recompute_infestation_derived = {
	clamp_variable = { var = iris_infestation_level min = 0 max = 100 }

	set_variable = { var = iris_infestation_bonus value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_bonus value = 0.005 }

	set_variable = { var = iris_infestation_supply value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_supply value = 0.02 }

	set_variable = { var = iris_infestation_supply_factor value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_supply_factor value = 0.01 }

	set_variable = { var = iris_infestation_enemy_supply value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_enemy_supply value = -0.005 }

	set_variable = { var = iris_infestation_biomass value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_biomass value = 0.05 }

	set_variable = { var = iris_infestation_penalty value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_penalty value = -0.005 }

	set_variable = { var = iris_infestation_attrition value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_attrition value = -0.002 }

	set_variable = { var = iris_infestation_suppression value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_suppression value = -0.008 }

	set_variable = { var = iris_infestation_decay value = iris_infestation_level }
	multiply_variable = { var = iris_infestation_decay value = 0.008 }
}

SRP_restamp_infestation_modifier = {
	if = {
		limit = {
			OR = {
				state_is_iris_bloc_controlled = yes
				is_controlled_by = KNX
			}
			check_variable = { iris_infestation_level > 0 }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = beckon_hostile_debuff } }
			remove_dynamic_modifier = { modifier = beckon_hostile_debuff }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = beckon_iris_buff } }
			remove_dynamic_modifier = { modifier = beckon_iris_buff }
		}
		add_dynamic_modifier = {
			modifier = beckon_iris_buff
			days = 2
		}
	}

	if = {
		limit = {
			NOT = {
				OR = {
					state_is_iris_bloc_controlled = yes
					is_controlled_by = KNX
				}
			}
			check_variable = { iris_infestation_level > 0 }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = beckon_iris_buff } }
			remove_dynamic_modifier = { modifier = beckon_iris_buff }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = beckon_hostile_debuff } }
			remove_dynamic_modifier = { modifier = beckon_hostile_debuff }
		}
		add_dynamic_modifier = {
			modifier = beckon_hostile_debuff
			days = 2
		}
	}

	if = {
		limit = {
			NOT = { check_variable = { iris_infestation_level > 0 } }
			has_dynamic_modifier = { modifier = beckon_iris_buff }
		}
		remove_dynamic_modifier = { modifier = beckon_iris_buff }
	}

	if = {
		limit = {
			NOT = { check_variable = { iris_infestation_level > 0 } }
			has_dynamic_modifier = { modifier = beckon_hostile_debuff }
		}
		remove_dynamic_modifier = { modifier = beckon_hostile_debuff }
	}
}

SRP_refresh_infestation_state = {
	SRP_recompute_infestation_derived = yes
	SRP_restamp_infestation_modifier = yes
}

	multiply_variable = { var = IRS_adapt_surface value = 0.02 }

	set_variable = { var = IRS_adapt_held value = IRS_lean_held }
	multiply_variable = { var = IRS_adapt_held value = 0.02 }

	set_variable = { var = IRS_adapt_loose value = IRS_lean_loose }
	multiply_variable = { var = IRS_adapt_loose value = -0.02 }

	set_variable = { var = IRS_adapt_centre value = IRS_lean_centre }
	multiply_variable = { var = IRS_adapt_centre value = 0.03 }

	set_variable = { var = IRS_adapt_thin value = IRS_lean_thin }
	multiply_variable = { var = IRS_adapt_thin value = 0.02 }

}

IRS_stamp_lineage_by_difficulty = {
	if = {
		limit = { NOT = { difficulty > 0 } }
		set_country_flag = IRS_archetype_tromos
		add_ideas = IRS_archetype_tromos_spirit
	}
	if = {
		limit = { difficulty > 0  NOT = { difficulty > 1 } }
		set_country_flag = IRS_archetype_vitreous
		add_ideas = IRS_archetype_vitreous_spirit
	}
	if = {
		limit = { difficulty > 1  NOT = { difficulty > 2 } }
		set_country_flag = IRS_archetype_iris
		add_ideas = IRS_archetype_iris_spirit
	}
	if = {
		limit = {
			difficulty > 2  NOT = { difficulty > 3 }
			has_country_flag = IRS_collapse_unchecked
		}
		set_country_flag = IRS_archetype_unchecked
		add_ideas = IRS_archetype_unchecked_spirit
	}
	if = {
		limit = {
			difficulty > 2  NOT = { difficulty > 3 }
			has_country_flag = IRS_collapse_adaptive
		}
		set_country_flag = IRS_archetype_adaptive
		add_ideas = IRS_archetype_adaptive_spirit
	}
	if = {
		limit = { difficulty > 3 }
		set_country_flag = IRS_archetype_telotic
		add_ideas = IRS_archetype_telotic_spirit
	}
	set_country_flag = IRS_lineage_resolved

	if = {
		limit = { has_country_flag = IRS_archetype_telotic }
		load_oob = "IRS_telotic_navy"
	}
}

IRS_resolve_lineage = {
	set_variable = { var = IRS_net_coord value = IRS_lean_held }
	subtract_from_variable = { var = IRS_net_coord value = IRS_lean_loose }

	set_variable = { var = IRS_net_centre value = IRS_lean_centre }
	subtract_from_variable = { var = IRS_net_centre value = IRS_lean_thin }

	set_variable = { var = IRS_net_depth value = IRS_lean_deep }
	subtract_from_variable = { var = IRS_net_depth value = IRS_lean_surface }

	if = {
		limit = { difficulty > 3 }
		set_country_flag = IRS_archetype_telotic
		add_ideas = IRS_archetype_telotic_spirit
		set_country_flag = IRS_lineage_resolved
	}
	if = {
		limit = {
			NOT = { has_country_flag = IRS_lineage_resolved }
			check_variable = { IRS_net_coord < 0 }
		}
		set_country_flag = IRS_archetype_unchecked
		add_ideas = IRS_archetype_unchecked_spirit
		set_country_flag = IRS_lineage_resolved
	}
	if = {
		limit = {
			NOT = { has_country_flag = IRS_lineage_resolved }
			check_variable = { IRS_net_centre < 1 }
		}
		set_country_flag = IRS_archetype_adaptive
		add_ideas = IRS_archetype_adaptive_spirit
		set_country_flag = IRS_lineage_resolved
	}
	if = {
		limit = {
			NOT = { has_country_flag = IRS_lineage_resolved }
			check_variable = { IRS_net_depth > 0 }
		}
		set_country_flag = IRS_archetype_vitreous
		add_ideas = IRS_archetype_vitreous_spirit
		set_country_flag = IRS_lineage_resolved
	}
	if = {
		limit = {
			NOT = { has_country_flag = IRS_lineage_resolved }
			check_variable = { IRS_net_depth < 0 }
		}
		set_country_flag = IRS_archetype_tromos
		add_ideas = IRS_archetype_tromos_spirit
		set_country_flag = IRS_lineage_resolved
	}
	if = {
		limit = { NOT = { has_country_flag = IRS_lineage_resolved } }
		set_country_flag = IRS_archetype_iris
		add_ideas = IRS_archetype_iris_spirit
		set_country_flag = IRS_lineage_resolved
	}
}

		add_research_slot = 1

			limit = { check_variable = { IRS_lean_surface > 1 } }

		}

			limit = { check_variable = { IRS_lean_held > 1 } }
