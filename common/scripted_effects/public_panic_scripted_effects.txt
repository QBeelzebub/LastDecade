SRP_public_panic_own_floor_effect = {
    if = {
        limit = { check_variable = { public_panic_peak_states < num_owned_controlled_states } }
        set_variable = { var = public_panic_peak_states value = num_owned_controlled_states }
    }
    set_variable = { var = public_panic_own_floor value = public_panic_peak_states }
    subtract_from_variable = { var = public_panic_own_floor value = num_owned_controlled_states }
    multiply_variable = { var = public_panic_own_floor value = 8 }

    set_variable = { public_panic_infested_states = 0 }
    every_controlled_state = {
        limit = { check_variable = { iris_infestation_level > 49 } }
        PREV = { add_to_variable = { public_panic_infested_states = 1 } }
    }
    multiply_variable = { var = public_panic_infested_states value = 5 }
    add_to_variable = { var = public_panic_own_floor value = public_panic_infested_states }

    if = {
        limit = {
            any_enemy_country = { is_full_parasite = yes }
        }
        add_to_variable = { var = public_panic_own_floor value = 50 }
    }
    clamp_variable = { var = public_panic_own_floor min = 0 max = 400 }
}


SRP_recalculate_public_panic_effect = {
    SRP_public_panic_tension_mult_effect = yes
    set_temp_variable = { panic_delta = public_panic }
    subtract_from_temp_variable = { panic_delta = public_panic_last }
    if = {
        limit = { check_variable = { panic_delta > 0 } }
        subtract_from_variable = { var = public_panic value = panic_delta }
        multiply_temp_variable = { panic_delta = public_panic_tension_mult }
        if = {
            limit = { has_global_flag = srp_panic_gains_low }
            multiply_temp_variable = { panic_delta = 0.5 }
        }
        if = {
            limit = { has_global_flag = srp_panic_gains_high }
            multiply_temp_variable = { panic_delta = 1.5 }
        }
        if = {
            limit = { has_country_flag = EFH_attenborough_in_office }
            multiply_temp_variable = { panic_delta = 0.9 }
        }
        add_to_variable = { var = public_panic value = panic_delta }
    }

    SRP_public_panic_tension_floor_effect = yes
    add_to_variable = { var = public_panic_floor value = public_panic_own_floor }
    if = {
        limit = { has_country_flag = EFH_lauterbach_in_office }
        add_to_variable = { var = public_panic_floor value = 50 }
    }
    if = {
        limit = { has_country_flag = EFH_armageddon_declared }
        set_variable = { var = public_panic_floor value = 900 }
    }
    if = {
        limit = { check_variable = { public_panic < public_panic_floor } }
        set_variable = { var = public_panic value = public_panic_floor }
    }

    clamp_variable = { var = public_panic min = 0 max = 1000 }
    set_variable = { var = public_panic_last value = public_panic }


    set_variable = { var = public_panic_stab_factor value = 0 }
    add_to_variable = { var = public_panic_stab_factor value = public_panic }
    multiply_variable = { var = public_panic_stab_factor value = -0.0002 }

    set_variable = { var = public_panic_pp_factor value = 0 }
    add_to_variable = { var = public_panic_pp_factor value = public_panic }
    multiply_variable = { var = public_panic_pp_factor value = -0.00015 }

    set_variable = { var = public_panic_ws_factor value = 0 }
    add_to_variable = { var = public_panic_ws_factor value = public_panic }
    multiply_variable = { var = public_panic_ws_factor value = -0.00012 }

    set_variable = { var = public_panic_above_threshold value = public_panic }
    add_to_variable = { var = public_panic_above_threshold value = -300 }
    clamp_variable = { var = public_panic_above_threshold min = 0 max = 700 }

    set_variable = { var = public_panic_research_factor value = 0 }
    add_to_variable = { var = public_panic_research_factor value = public_panic_above_threshold }
    multiply_variable = { var = public_panic_research_factor value = 0.001 }
    multiply_variable = { var = public_panic_research_factor value = 0.2143 }

    set_variable = { var = public_panic_production_factor value = 0 }
    add_to_variable = { var = public_panic_production_factor value = public_panic_above_threshold }
    multiply_variable = { var = public_panic_production_factor value = 0.001 }
    multiply_variable = { var = public_panic_production_factor value = 0.1429 }

    set_variable = { var = public_panic_manpower_factor value = 0 }
    add_to_variable = { var = public_panic_manpower_factor value = public_panic_above_threshold }
    multiply_variable = { var = public_panic_manpower_factor value = 0.5 }

    set_variable = { var = public_panic_resistance_factor value = 0 }
    add_to_variable = { var = public_panic_resistance_factor value = public_panic_above_threshold }
    multiply_variable = { var = public_panic_resistance_factor value = 0.001 }
    multiply_variable = { var = public_panic_resistance_factor value = 4.286 }

    if = {
        limit = { has_dynamic_modifier = { modifier = public_panic_modifier } }
        remove_dynamic_modifier = { modifier = public_panic_modifier }
    }
    add_dynamic_modifier = { modifier = public_panic_modifier }
}
