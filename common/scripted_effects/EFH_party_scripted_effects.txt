		limit = { NOT = { has_country_flag = EFH_party_system_initialized } }

		set_variable = { var = EFH_party_volt value = 15 }

EFH_party_clamp_support_effect = {
	clamp_variable = { var = EFH_party_astra min = 0 max = 100 }
	clamp_variable = { var = EFH_party_volt min = 0 max = 100 }
	clamp_variable = { var = EFH_party_ave min = 0 max = 100 }
	clamp_variable = { var = EFH_party_continuity min = 0 max = 100 }
	clamp_variable = { var = EFH_party_social_green min = 0 max = 100 }
	clamp_variable = { var = EFH_party_sovereigntist min = 0 max = 100 }
	clamp_variable = { var = EFH_separatist_pressure min = 0 max = 100 }

	set_variable = { var = EFH_party_total value = 0 }
	add_to_variable = { var = EFH_party_total value = EFH_party_astra }
	add_to_variable = { var = EFH_party_total value = EFH_party_volt }
	add_to_variable = { var = EFH_party_total value = EFH_party_ave }
	add_to_variable = { var = EFH_party_total value = EFH_party_continuity }
	add_to_variable = { var = EFH_party_total value = EFH_party_social_green }
	add_to_variable = { var = EFH_party_total value = EFH_party_sovereigntist }

	if = {
		limit = { check_variable = { var = EFH_party_total value = 0 compare = less_than_or_equals } }
		set_variable = { var = EFH_party_astra value = 12 }
		set_variable = { var = EFH_party_volt value = 15 }
		set_variable = { var = EFH_party_ave value = 8 }
		set_variable = { var = EFH_party_continuity value = 40 }
		set_variable = { var = EFH_party_social_green value = 18 }
		set_variable = { var = EFH_party_sovereigntist value = 7 }
	}
	else = {
		multiply_variable = { var = EFH_party_astra value = 100 }
		divide_variable = { var = EFH_party_astra value = EFH_party_total }
		multiply_variable = { var = EFH_party_volt value = 100 }
		divide_variable = { var = EFH_party_volt value = EFH_party_total }
		multiply_variable = { var = EFH_party_ave value = 100 }
		divide_variable = { var = EFH_party_ave value = EFH_party_total }
		multiply_variable = { var = EFH_party_continuity value = 100 }
		divide_variable = { var = EFH_party_continuity value = EFH_party_total }
		multiply_variable = { var = EFH_party_social_green value = 100 }
		divide_variable = { var = EFH_party_social_green value = EFH_party_total }
		multiply_variable = { var = EFH_party_sovereigntist value = 100 }
		divide_variable = { var = EFH_party_sovereigntist value = EFH_party_total }
	}
}

EFH_party_check_separatist_thresholds_effect = {
	if = {
		limit = {
			check_variable = { var = EFH_separatist_pressure value = 40 compare = greater_than_or_equals }
			NOT = { has_country_flag = EFH_separatist_unrest_warned }
		}
		set_country_flag = EFH_separatist_unrest_warned
		add_ideas = EFH_separatist_unrest
		news_event = { id = efh_party.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { var = EFH_separatist_pressure value = 60 compare = greater_than_or_equals }
			NOT = { has_country_flag = EFH_restoration_referendum_warned }
		}
		set_country_flag = EFH_restoration_referendum_warned
		add_ideas = EFH_restoration_referendum_crisis
		news_event = { id = efh_party.21 days = 1 }
	}
	if = {
		limit = {
			check_variable = { var = EFH_separatist_pressure value = 80 compare = greater_than_or_equals }
			has_country_flag = EFH_party_sovereigntist_incumbent
			NOT = { has_country_flag = EFH_dissolution_congress_fired }
		}
		set_country_flag = EFH_dissolution_congress_fired
		country_event = { id = efh_party.30 days = 1 }
	}
	if = {
		limit = {
			check_variable = { var = EFH_separatist_pressure value = 40 compare = less_than }
			has_country_flag = EFH_separatist_unrest_warned
		}
		clr_country_flag = EFH_separatist_unrest_warned
		remove_ideas = EFH_separatist_unrest
		news_event = { id = efh_party.22 days = 1 }
	}
	if = {
		limit = {
			check_variable = { var = EFH_separatist_pressure value = 60 compare = less_than }
			has_country_flag = EFH_restoration_referendum_warned
		}
		clr_country_flag = EFH_restoration_referendum_warned
		remove_ideas = EFH_restoration_referendum_crisis
		news_event = { id = efh_party.23 days = 1 }
	}
}
