LD_update_beckon_globe = {
	set_variable = { global.LD_world_states = 0 }
	set_variable = { global.LD_hive_states = 0 }
	every_state = {
		add_to_variable = { global.LD_world_states = 1 }
	}
	if = {
		limit = { has_global_flag = parasites_emerged }
		every_state = {
			limit = {
				OR = {
					controller = { is_full_parasite = yes }
					check_variable = { iris_infestation_level > 49 }
				}
			}
			add_to_variable = { global.LD_hive_states = 1 }
		}
	}

	set_variable = { global.LD_beckon_globe_frame = 1 }
	if = { limit = { check_variable = { global.LD_hive_states > 0 } } set_variable = { global.LD_beckon_globe_frame = 2 } }
	if = { limit = { check_variable = { global.LD_hive_states > 109 } } set_variable = { global.LD_beckon_globe_frame = 3 } }
	if = { limit = { check_variable = { global.LD_hive_states > 253 } } set_variable = { global.LD_beckon_globe_frame = 4 } }
	if = { limit = { check_variable = { global.LD_hive_states > 473 } } set_variable = { global.LD_beckon_globe_frame = 5 } }
	if = { limit = { check_variable = { global.LD_hive_states > 748 } } set_variable = { global.LD_beckon_globe_frame = 6 } }
	if = { limit = { check_variable = { global.LD_hive_states > 949 } } set_variable = { global.LD_beckon_globe_frame = 7 } }
	if = { limit = { check_variable = { global.LD_hive_states > 1011 } } set_variable = { global.LD_beckon_globe_frame = 8 } }
	if = { limit = { check_variable = { global.LD_hive_states > 1015 } } set_variable = { global.LD_beckon_globe_frame = 9 } }
	if = { limit = { check_variable = { global.LD_hive_states > 1041 } } set_variable = { global.LD_beckon_globe_frame = 10 } }
}
