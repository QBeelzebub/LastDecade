
SRP_absorb_state_for_IRS = {
	add_core_of = IRS
	every_country = {
		limit = { NOT = { is_full_parasite = yes } }
		PREV = { remove_core_of = PREV }
	}
	add_dynamic_modifier = { modifier = irs_consumed_state }
	set_state_flag = irs_state_consumed
	if = {
		limit = { industrial_complex > 0 }
		remove_building = { type = industrial_complex level = 20 }
	}
	if = {
		limit = { arms_factory > 0 }
		remove_building = { type = arms_factory level = 20 }
	}
	if = {
		limit = { infrastructure > 0 }
		remove_building = { type = infrastructure level = 5 }
	}
	if = {
		limit = { dockyard > 0 }
		remove_building = { type = dockyard level = 20 }
	}
	if = {
		limit = { synthetic_refinery > 0 }
		remove_building = { type = synthetic_refinery level = 3 }
	}
	if = {
		limit = { fuel_silo > 0 }
		remove_building = { type = fuel_silo level = 20 }
	}
	if = {
		limit = { radar_station > 0 }
		remove_building = { type = radar_station level = 6 }
	}
	if = {
		limit = { rocket_site > 0 }
		remove_building = { type = rocket_site level = 3 }
	}
	if = {
		limit = { energy_infrastructure > 0 }
		remove_building = { type = energy_infrastructure level = 1 }
	}
	if = {
		limit = { industrial_infrastructure > 0 }
		remove_building = { type = industrial_infrastructure level = 1 }
	}
	add_building_construction = {
		type = beckon_cluster
		level = 1
	}
	IRS = {
		set_variable = { var = IRS_ep_gain_tmp value = 25 }
		IRS_grant_scaled_ep = yes
	}
}

SRP_absorb_state_for_KNX = {
	remove_core_of = IRS
	remove_dynamic_modifier = { modifier = irs_consumed_state }
	clr_state_flag = irs_state_consumed
	add_core_of = KNX
	every_country = {
		limit = { NOT = { is_full_parasite = yes } }
		PREV = { remove_core_of = PREV }
	}
	add_dynamic_modifier = { modifier = knx_consumed_state }
	set_state_flag = knx_state_consumed
}
